2. ALL POSTGRESQL DATABASE TABLES & SCHEMA DIRECTORY
================================================================================
The "barangay_drrm" database contains 7 public tables:

--------------------------------------------------------------------------------
TABLE 1: households
Model:   App\Models\Household
Purpose: Masterlist of registered households, alert statuses, family members.
Columns:
  * id                    VARCHAR(50) PRIMARY KEY (e.g., 'hh-buang-1')
  * head_of_family        VARCHAR(150) NOT NULL
  * birthday              DATE NULL
  * age                   INTEGER NOT NULL
  * contact_number        VARCHAR(50) NOT NULL
  * number_of_members     INTEGER NOT NULL DEFAULT 1
  * house_number          VARCHAR(50) NULL
  * purok                 VARCHAR(50) NOT NULL
  * barangay              VARCHAR(50) NOT NULL ('Buang' or 'Nagsipit')
  * complete_address      TEXT NOT NULL
  * location              JSONB NOT NULL DEFAULT '{"lat": 0.0, "lng": 0.0}'
  * status                VARCHAR(20) NOT NULL ('SAFE', 'MODERATE', 'EMERGENCY')
  * status_note           TEXT NULL
  * evacuation_center_id  VARCHAR(50) NULL (FK -> evacuation_centers.id)
  * verification_status   VARCHAR(20) NOT NULL ('Pending' or 'Verified')
  * special_needs         JSONB NOT NULL DEFAULT '[]' (Senior, Infant, PWD, Pregnant)
  * family_members        JSONB NOT NULL DEFAULT '[]' (Array of family member objects)
  * last_status_update    TIMESTAMP WITH TIME ZONE
  * created_at            TIMESTAMP WITH TIME ZONE
  * updated_at            TIMESTAMP WITH TIME ZONE

--------------------------------------------------------------------------------
TABLE 2: evacuation_centers
Model:   App\Models\EvacuationCenter
Purpose: Designated disaster evacuation sites, live capacities and facilities.
Columns:
  * id                    VARCHAR(50) PRIMARY KEY (e.g., 'center-buang-1')
  * name                  VARCHAR(150) NOT NULL
  * barangay              VARCHAR(50) NOT NULL ('Buang' or 'Nagsipit')
  * purok                 VARCHAR(50) NOT NULL
  * capacity_families     INTEGER NOT NULL DEFAULT 0
  * assigned_count        INTEGER NOT NULL DEFAULT 0
  * location              JSONB NOT NULL DEFAULT '{"lat": 0.0, "lng": 0.0}'
  * manager_name          VARCHAR(150) NOT NULL
  * manager_contact       VARCHAR(50) NOT NULL
  * status                VARCHAR(20) NOT NULL ('Open', 'Full', 'Standby')
  * facilities            JSONB NOT NULL DEFAULT '[]' (Generator, Clean Water, etc.)
  * created_at            TIMESTAMP WITH TIME ZONE
  * updated_at            TIMESTAMP WITH TIME ZONE

--------------------------------------------------------------------------------
TABLE 3: relief_distributions
Model:   App\Models\ReliefDistribution
Purpose: Disaster goods, food packs, and relief supply distribution records.
Columns:
  * id                    VARCHAR(50) PRIMARY KEY (e.g., 'relief-6705...')
  * household_id          VARCHAR(50) NOT NULL (FK -> households.id)
  * recipient_name        VARCHAR(150) NOT NULL
  * barangay              VARCHAR(50) NOT NULL ('Buang' or 'Nagsipit')
  * purok                 VARCHAR(50) NOT NULL
  * item                  VARCHAR(150) NOT NULL
  * qty                   VARCHAR(50) NOT NULL
  * location              VARCHAR(150) NOT NULL
  * distributed_at        TIMESTAMP WITH TIME ZONE
  * officer_name          VARCHAR(150) NOT NULL
  * notes                 TEXT NULL
  * created_at            TIMESTAMP WITH TIME ZONE
  * updated_at            TIMESTAMP WITH TIME ZONE

--------------------------------------------------------------------------------
TABLE 4: official_accounts
Model:   App\Models\OfficialAccount
Purpose: Authorized barangay officials and DRRM officers login accounts.
Columns:
  * id                    SERIAL PRIMARY KEY
  * username              VARCHAR(100) UNIQUE NOT NULL (e.g., 'Buangadmin.gov')
  * password_hash         VARCHAR(255) NOT NULL (Bcrypt hashed)
  * barangay              VARCHAR(50) NOT NULL ('Buang' or 'Nagsipit')
  * official_name         VARCHAR(150) NOT NULL
  * position              VARCHAR(150) NOT NULL
  * phone                 VARCHAR(50) NOT NULL
  * security_question     TEXT NOT NULL
  * security_answer       TEXT NOT NULL
  * created_at            TIMESTAMP WITH TIME ZONE
  * updated_at            TIMESTAMP WITH TIME ZONE

--------------------------------------------------------------------------------
TABLE 5: activity_logs
Model:   App\Models\ActivityLog
Purpose: Audit trail of operational activities, logins, and status updates.
Columns:
  * id                    VARCHAR(60) PRIMARY KEY
  * timestamp             TIMESTAMP WITH TIME ZONE NOT NULL
  * action_type           VARCHAR(50) NOT NULL ('LOGIN', 'STATUS_UPDATE', etc.)
  * actor_name            VARCHAR(150) NOT NULL
  * actor_role            VARCHAR(100) NOT NULL
  * barangay              VARCHAR(50) NOT NULL ('Buang' or 'Nagsipit')
  * target                VARCHAR(150) NOT NULL
  * details               TEXT NOT NULL
  * is_working_hours      BOOLEAN NOT NULL DEFAULT true
  * work_shift_tag        VARCHAR(50) NOT NULL
  * terminal              VARCHAR(100) DEFAULT 'Main Barangay Hall Terminal'
  * created_at            TIMESTAMP WITH TIME ZONE

--------------------------------------------------------------------------------
TABLE 6: personal_access_tokens
Purpose: Laravel Sanctum API authentication tokens for secure REST sessions.
Columns:
  * id                    BIGSERIAL PRIMARY KEY
  * tokenable_type        VARCHAR(255) NOT NULL
  * tokenable_id          BIGINT NOT NULL
  * name                  VARCHAR(255) NOT NULL
  * token                 VARCHAR(64) UNIQUE NOT NULL
  * abilities             TEXT NULL
  * last_used_at          TIMESTAMP WITH TIME ZONE
  * expires_at            TIMESTAMP WITH TIME ZONE
  * created_at            TIMESTAMP WITH TIME ZONE
  * updated_at            TIMESTAMP WITH TIME ZONE

--------------------------------------------------------------------------------
TABLE 7: migrations
Purpose: Internal Laravel system table tracking applied database migrations.
Columns:
  * id                    SERIAL PRIMARY KEY
  * migration             VARCHAR(255) NOT NULL
  * batch                 INTEGER NOT NULL

3. HOW TO VIEW DATABASE NAME & TABLES USING COMMANDS
================================================================================

--- METHOD A: Using Laravel Artisan (Easiest) ---
Run in: c:\Users\King\Veryupdated thesis\laravel-backend

1. View all table names in one quick command:
    php artisan tinker --execute="dump(Illuminate\Support\Facades\Schema::getTableListing());"

2. View database name, host, and port:
    php artisan db:show

3. View detailed columns of a specific table:
    php artisan db:table households
    php artisan db:table evacuation_centers
    php artisan db:table relief_distributions
    php artisan db:table official_accounts
    php artisan db:table activity_logs


--- METHOD B: Using PostgreSQL Command Line (psql) ---
Run from PowerShell:

1. List all databases in PostgreSQL:
    & "C:\Program Files\PostgreSQL\18\bin\psql.exe" -U postgres -c "\l"

2. List all tables in the "barangay_drrm" database:
    & "C:\Program Files\PostgreSQL\18\bin\psql.exe" -U postgres -d barangay_drrm -c "\dt"

3. Inspect column definitions of the "households" table:
    & "C:\Program Files\PostgreSQL\18\bin\psql.exe" -U postgres -d barangay_drrm -c "\d+ households"

4. Query all table names via SQL:
    & "C:\Program Files\PostgreSQL\18\bin\psql.exe" -U postgres -d barangay_drrm -c "SELECT table_name FROM information_schema.tables WHERE table_schema = 'public' ORDER BY table_name;"

2. Run any new pending migrations:

    php artisan db:show

7. HOUSEHOLD CRUD GUIDE (via php artisan tinker)
================================================================================
First, enter the interactive shell:
    cd "c:\Users\King\Veryupdated thesis\laravel-backend"
    php artisan tinker

--- R: READ (VIEW DATA) ---

1. View all household IDs and Head of Family names:
    App\Models\Household::pluck('head_of_family', 'id');

2. View key columns of all households:
    App\Models\Household::all(['id', 'head_of_family', 'barangay', 'purok', 'status', 'contact_number'])->toArray();

3. View a single household by ID:
    $h = App\Models\Household::find('hh-buang-1');
    echo $h->head_of_family . " | " . $h->status . " | " . $h->contact_number;

4. View individual family members inside a household:
    $h = App\Models\Household::find('hh-buang-1');
    $h->family_members;

5. View special needs tags (Senior, Infant, PWD, Pregnant):
    $h = App\Models\Household::find('hh-buang-1');
    $h->special_needs;

6. Filter households by Barangay ('Buang' or 'Nagsipit'):
    App\Models\Household::where('barangay', 'Buang')->get(['id', 'head_of_family', 'purok']);
    App\Models\Household::where('barangay', 'Nagsipit')->get(['id', 'head_of_family', 'purok']);

7. Filter households by Alert Status ('EMERGENCY', 'MODERATE', 'SAFE'):
    App\Models\Household::where('status', 'EMERGENCY')->get(['id', 'head_of_family', 'status_note']);
    App\Models\Household::where('status', 'SAFE')->get(['id', 'head_of_family']);

8. Count households:
    App\Models\Household::count();
    App\Models\Household::where('status', 'EMERGENCY')->count();
    App\Models\Household::where('verification_status', 'Verified')->count();

--- U: UPDATE (MODIFY AN EXISTING HOUSEHOLD) ---

1. Change Alert Status (SAFE -> EMERGENCY):
    $h = App\Models\Household::find('hh-buang-1');
    $h->update([
        'status' => 'SAFE',
        'status_note' => 'Floodwaters receded; situation stabilized.',
        'last_status_update' => now(),
    ]);

2. Update contact number or address:
    $h = App\Models\Household::find('hh-buang-1');
    $h->contact_number = '0999-888-7777';
    $h->complete_address = 'Updated Address, Purok 1, Buang';
    $h->save();

3. Assign household to an Evacuation Center:
    $h = App\Models\Household::find('hh-buang-1');
    $h->evacuation_center_id = 'center-buang-1';
    $h->save();
    // Update live count on evacuation center:
    App\Models\EvacuationCenter::find('center-buang-1')->recalculateAssignedCount();

4. Approve / Verify a household:
    $h = App\Models\Household::find('hh-buang-1');
    $h->update(['verification_status' => 'Verified']);


--- D: DELETE (REMOVE A HOUSEHOLD) ---

1. Delete by ID directly:
    App\Models\Household::destroy('hh-buang-2');

2. Delete using instance:
    $h = App\Models\Household::find('hh-buang-2');
    if ($h) { $h->delete(); echo "Deleted"; }
